**NAT 끄기 (작업 종료, `daily-stop.sh` 다음)**
```bash
#!/usr/bin/env bash
# End of day, AFTER scripts/daily-stop.sh: delete both NAT gateways and release their EIPs.
set -euo pipefail
export AWS_REGION=ap-northeast-2 AWS_DEFAULT_REGION=ap-northeast-2
nats=$(aws ec2 describe-nat-gateways --filter Name=tag:Project,Values=petclinic Name=state,Values=available,pending \
  --query 'NatGateways[].NatGatewayId' --output text)
[ -z "$nats" ] && { echo "no NAT gateways"; } || {
  for n in $nats; do aws ec2 delete-nat-gateway --nat-gateway-id "$n" >/dev/null; echo "deleting $n"; done
  aws ec2 wait nat-gateway-deleted --nat-gateway-ids $nats; }
for a in $(aws ec2 describe-addresses --filters Name=tag:Project,Values=petclinic --query 'Addresses[].AllocationId' --output text); do
  aws ec2 release-address --allocation-id "$a"; echo "released $a"
done
```

**NAT 켜기 (작업 시작, `daily-start.sh` 전)**
```bash
#!/usr/bin/env bash
# Start of day, BEFORE scripts/daily-start.sh: one EIP + NAT per AZ, app route tables pointed at them.
set -euo pipefail
export AWS_REGION=ap-northeast-2 AWS_DEFAULT_REGION=ap-northeast-2
P=petclinic
tag() { echo "ResourceType=$1,Tags=[{Key=Name,Value=$2},{Key=Project,Value=$P}]"; }
id() { aws ec2 "describe-$1" --filters "Name=tag:Name,Values=$2" --query "$3" --output text; }
for az in a c; do
  if [ "$(aws ec2 describe-nat-gateways --filter "Name=tag:Name,Values=$P-nat-2$az" Name=state,Values=available,pending \
      --query 'length(NatGateways)')" != 0 ]; then echo "$P-nat-2$az exists"; continue; fi
  pub=$(id subnets "$P-public-2$az" 'Subnets[0].SubnetId')
  rt=$(id route-tables "$P-rt-app-2$az" 'RouteTables[0].RouteTableId')
  eip=$(aws ec2 allocate-address --domain vpc --tag-specifications "$(tag elastic-ip "$P-nat-2$az")" --query AllocationId --output text)
  nat=$(aws ec2 create-nat-gateway --subnet-id "$pub" --allocation-id "$eip" \
    --tag-specifications "$(tag natgateway "$P-nat-2$az")" --query NatGateway.NatGatewayId --output text)
  aws ec2 wait nat-gateway-available --nat-gateway-ids "$nat"
  aws ec2 replace-route --route-table-id "$rt" --destination-cidr-block 0.0.0.0/0 --nat-gateway-id "$nat"
  echo "$P-nat-2$az up"
```
