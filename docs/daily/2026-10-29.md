  ```bash
  ./scripts/teardown-all.sh --dry-run --keep-dns

  kubectl -n petclinic patch scaledobject/<was-stable|was-canary> --type merge -p '{"metadata":{"finalizers":null}}'
  # Route 53 레코드·영역, cap-shop TG·EBS·AMI·스냅샷·VPC 삭제 (스크래치 스크립트, ID 고정)
  ```
